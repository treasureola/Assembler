# INSTRUCTION SEGMENT 
LOC 30

# I/O TESTS
IN 0,0          # Input a number into R0
STR 0,1,8       # Store input to memory[8] (using IX1)
OUT 0,1         # Output the same value

# LOAD / STORE / INDEX REGISTER TESTS
LDR 1,0,7       # Load memory[7] into R1
STR 1,0,9       # Store R1 into memory[9]
LDX 1,7         # Load IX1 with memory[7]
STX 1,10        # Store IX1 back to memory[10]

# ARITHMETIC TESTS
AIR 1,5         # R1 = R1 + 5
SIR 1,3         # R1 = R1 - 3
LDR 2,0,6       # Load R2 with memory[6] (value 3)
AMR 2,0,7       # R2 = R2 + memory[7]
SMR 2,0,8       # R2 = R2 - memory[8]
STR 2,0,11      # Store R2 to memory[11]

# LOGICAL TESTS
LDR 3,0,9       # Load R3
NOT 3           # Bitwise NOT
AND 3,0,6       # R3 = R3 AND memory[6]
ORR 3,0,7       # R3 = R3 OR memory[7]
STR 3,0,12      # Store R3 to memory[12]

#SHIFT TESTS 
LDR 1,0,12
SRC 1,1,0,1     # Shift right R1 by 1
RRC 1,1,1,0     # Rotate left R1 by 1
STR 1,0,13

# COMPARISON / CONTROL FLOW TESTS
LDR 2,0,11
LDR 3,0,13
TRR 2,3         # Compare R2 R3
JZ 0,50         # Jump if equal
JNE 0,60        # Jump if not equal
# SETCCE 1        # Set CC=1
HLT 0           # Should not reach here if JNE taken

#BRANCH TARGETS
LOC 50
LDR 0,0,6
OUT 0,1         # Output value for equality branch
JNE 0,70        # Jump to end

LOC 60
LDR 0,0,7
OUT 0,1         # Output value for non-equal branch

#CACHE TEST

LOC 70
LDR 0,0,100     # likely miss, load to cache

LDR 2,0,200     # miss, new line
LDR 3,0,100     # hit again

LDR 1,0,200     # may hit if still in cache
LDR 2,0,400     # miss, test cache overflow
